# =============================================================================
# External tools (apt / go / pipx / git)
# These are NOT installed by pip. Run the commands below on the box that
# will execute the authorized external test. HerecyEx detects what is
# present at startup and skips anything missing (with an install hint).
#
# The script never exploits, password-brutes, or runs DoS checks. Install
# only what you are comfortable running in an authorized external assessment.
# =============================================================================
#
# --- 0. Prerequisites -------------------------------------------------------
# sudo apt update
# sudo apt install -y git curl build-essential
# # Go 1.21+ (Kali usually already has it). Confirm with: go version
# # If go is missing:
# #   sudo apt install -y golang-go
# mkdir -p "$HOME/go/bin" "$HOME/.local/bin"
# echo 'export PATH="$PATH:$HOME/go/bin:$HOME/.local/bin:$HOME/.pdtm/go/bin"' >> ~/.zshrc
# export PATH="$PATH:$HOME/go/bin:$HOME/.local/bin:$HOME/.pdtm/go/bin"
#
# --- 1. Kali apt packages (core + useful extras) ----------------------------
# sudo apt install -y \
#   whois dnsutils curl nmap ncat \
#   theharvester dnsrecon \
#   amass massdns \
#   fping masscan rustscan \
#   whatweb wafw00f nikto \
#   gobuster feroxbuster ffuf \
#   eyewitness sslscan testssl.sh \
#   enum4linux enum4linux-ng smbmap smbclient \
#   onesixtyone snmp ldap-utils \
#   ssh-audit \
#   seclists
#
# Wordlists used for optional content discovery land in:
#   /usr/share/seclists/Discovery/Web-Content/raft-small-words.txt
#   /usr/share/seclists/Discovery/Web-Content/common.txt
#   /usr/share/wordlists/dirb/common.txt
#
# --- 2. ProjectDiscovery tools (strongly recommended) -----------------------
# Fastest path — pdtm installs/updates the whole suite:
#   go install github.com/projectdiscovery/pdtm/cmd/pdtm@latest
#   pdtm -ia subfinder,dnsx,httpx,naabu,nuclei,katana,tlsx
#   nuclei -update-templates
#
# Or install one-by-one:
#   go install github.com/projectdiscovery/subfinder/v2/cmd/subfinder@latest
#   go install github.com/projectdiscovery/dnsx/cmd/dnsx@latest
#   go install github.com/projectdiscovery/httpx/cmd/httpx@latest
#   go install github.com/projectdiscovery/naabu/v2/cmd/naabu@latest
#   go install github.com/projectdiscovery/nuclei/v3/cmd/nuclei@latest
#   go install github.com/projectdiscovery/katana/cmd/katana@latest
#   go install github.com/projectdiscovery/tlsx/cmd/tlsx@latest
#   nuclei -update-templates
#
# Why these:
#   subfinder  — passive subdomain discovery
#   dnsx       — fast, structured DNS resolution
#   httpx      — live HTTP/S probe (title, tech, TLS, status)
#   naabu      — fast TCP port front-end that feeds nmap -sV -sC
#   nuclei     — template checks; HerecyEx rate-limits and excludes
#                dos,fuzz,intrusive,brute tags
#   katana     — light crawl, --full only
#   tlsx       — batched TLS metadata (expiry, self-signed, mismatch)
#
# httpx name collision: the Python httpx CLI does not support -l / -tech-detect.
# HerecyEx ignores that binary and looks for ProjectDiscovery httpx in
# ~/go/bin, ~/.pdtm/go/bin, and ~/.local/bin first.
#
# naabu SYN scan may need libpcap and/or root:
#   sudo apt install -y libpcap-dev
#
# --- 3. Other Go / GitHub tools ---------------------------------------------
#   go install github.com/owasp-amass/amass/v4/cmd/amass@master
#       # amass is also available via apt; either works. Passive mode only.
#   go install github.com/tomnomnom/assetfinder@latest
#       # Additional passive subdomain source
#   go install github.com/tomnomnom/waybackurls@latest
#       # Historical URLs from the Wayback Machine (passive)
#   go install github.com/lc/gau/v2/cmd/gau@latest
#       # GetAllURLs — aggregated historical URLs (passive)
#   go install github.com/sensepost/gowitness@latest
#       # Headless Chrome screenshots of live URLs
#       # Requires a Chrome/Chromium browser:
#       #   sudo apt install -y chromium
#
# --- 4. rustscan / masscan (fast port front-ends, optional) -----------------
# Prefer naabu when present. If you want rustscan from GitHub instead of apt:
#   # https://github.com/RustScan/RustScan/releases
#   # Example (adjust tag / arch):
#   # curl -sL https://github.com/RustScan/RustScan/releases/latest/download/rustscan_amd64.deb -o /tmp/rustscan.deb
#   # sudo apt install -y /tmp/rustscan.deb
#
# masscan (if not already installed):
#   sudo apt install -y masscan
#   # Used only when root or passwordless sudo is available, at a conservative
#   # --rate 500 suitable for authorized external work.
#
# --- 5. testssl.sh / ssh-audit / enum4linux-ng (git / pipx fallbacks) -------
# Kali apt usually has these; fallbacks if a package is missing:
#
#   git clone --depth 1 https://github.com/drwetter/testssl.sh.git "$HOME/testssl.sh"
#   ln -sf "$HOME/testssl.sh/testssl.sh" "$HOME/.local/bin/testssl.sh"
#   chmod +x "$HOME/testssl.sh/testssl.sh"
#
#   python3 -m pipx install ssh-audit
#   # or: sudo apt install -y ssh-audit
#
#   git clone --depth 1 https://github.com/cddmp/enum4linux-ng.git "$HOME/enum4linux-ng"
#   python3 -m pip install --user -r "$HOME/enum4linux-ng/requirements.txt"
#   ln -sf "$HOME/enum4linux-ng/enum4linux-ng.py" "$HOME/.local/bin/enum4linux-ng"
#   chmod +x "$HOME/enum4linux-ng/enum4linux-ng.py"
#
# --- 6. Quick verify --------------------------------------------------------
#   python3 external_recon_workflow.py --check-tools
#

# reports OK. Missing optional tools are skipped at runtime, not fatal.
